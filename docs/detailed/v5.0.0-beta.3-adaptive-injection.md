# Adaptive Injection Strategy — v5.0.0-beta.3 详细设计

## 架构

```
User Request → TaskTypeDetector → taskType
    → AdaptiveInjector.selectStrategy(taskType)
    → InjectionStrategy (aggressive/balanced/minimal/contextual)
    → Context Engine injected
```

## 1. TaskTypeDetector

```typescript
type TaskType = "coding" | "review" | "debugging" | "planning" | "question" | "conversation" | "unknown";

interface TaskSignature {
  type: TaskType;
  confidence: number;
  keywords: string[];
}

class TaskTypeDetector {
  private history: Map<string, number> = new Map(); // taskType → occurrence count

  detect(input: string, fileContext?: string[]): TaskSignature {
    const lower = input.toLowerCase();

    const rules: Array<{ type: TaskType; keywords: string[]; weight: number }> = [
      { type: "coding", keywords: ["implement", "create", "write code", "build", "add function"], weight: 1 },
      { type: "review", keywords: ["review", "check", "audit", "inspect", "examine"], weight: 1 },
      { type: "debugging", keywords: ["debug", "fix bug", "error", "crash", "broken", "issue"], weight: 1 },
      { type: "planning", keywords: ["plan", "design", "architect", "roadmap", "strategy"], weight: 1 },
      { type: "question", keywords: ["how to", "what is", "explain", "why", "when should"], weight: 0.8 },
      { type: "conversation", keywords: ["discuss", "talk about", "opinion", "thoughts"], weight: 0.7 },
    ];

    let best: TaskSignature = { type: "unknown", confidence: 0, keywords: [] };
    for (const rule of rules) {
      const hits = rule.keywords.filter((kw) => lower.includes(kw));
      if (hits.length > 0) {
        const confidence = Math.min(0.95, hits.length * 0.25 * rule.weight);
        if (confidence > best.confidence) best = { type: rule.type, confidence, keywords: hits };
      }
    }

    // Boost with file context
    if (fileContext?.length && best.type === "unknown") {
      const hasTests = fileContext.some((f) => /test|spec/.test(f));
      if (hasTests) best = { type: "coding", confidence: 0.6, keywords: ["file_context"] };
    }

    this.history.set(best.type, (this.history.get(best.type) ?? 0) + 1);
    return best;
  }
}
```

## 2. InjectionStrategy

```typescript
type StrategyMode = "aggressive" | "balanced" | "minimal" | "contextual";

interface StrategyConfig {
  mode: StrategyMode;
  maxTokens: number;
  topK: number;
  includeMemory: boolean;
  includeGovernance: boolean;
  includeCI: boolean;
  includeCrossDomain: boolean;
}

const STRATEGY_MAP: Record<TaskType, StrategyConfig> = {
  coding:      { mode: "aggressive", maxTokens: 4000, topK: 15, includeMemory: true, includeGovernance: true, includeCI: false, includeCrossDomain: false },
  review:      { mode: "balanced", maxTokens: 3000, topK: 10, includeMemory: true, includeGovernance: true, includeCI: true, includeCrossDomain: true },
  debugging:   { mode: "contextual", maxTokens: 5000, topK: 20, includeMemory: true, includeGovernance: false, includeCI: true, includeCrossDomain: false },
  planning:    { mode: "aggressive", maxTokens: 4000, topK: 12, includeMemory: true, includeGovernance: true, includeCI: false, includeCrossDomain: true },
  question:    { mode: "minimal", maxTokens: 1500, topK: 5, includeMemory: true, includeGovernance: false, includeCI: false, includeCrossDomain: false },
  conversation:{ mode: "balanced", maxTokens: 2000, topK: 8, includeMemory: true, includeGovernance: false, includeCI: false, includeCrossDomain: false },
  unknown:     { mode: "balanced", maxTokens: 2500, topK: 10, includeMemory: true, includeGovernance: false, includeCI: false, includeCrossDomain: false },
};
```

## 3. AdaptiveInjector

```typescript
interface InjectionConfig {
  strategy: StrategyConfig;
  adjustedMaxTokens: number;
  taskType: TaskType;
  confidence: number;
}

class AdaptiveInjector {
  private detector = new TaskTypeDetector();

  selectStrategy(taskType: TaskType): StrategyConfig {
    return STRATEGY_MAP[taskType] ?? STRATEGY_MAP.unknown;
  }

  adjustParameters(strategy: StrategyConfig, context: { budget?: number; urgency?: string }): InjectionConfig {
    let adjusted = strategy.maxTokens;
    if (context.budget) adjusted = Math.min(adjusted, context.budget);
    if (context.urgency === "high") adjusted = Math.min(adjusted * 1.5, 8000);

    return {
      strategy,
      adjustedMaxTokens: adjusted,
      taskType: "unknown",
      confidence: 0.5,
    };
  }

  async inject(input: string, context?: { budget?: number; urgency?: string; fileContext?: string[] }): Promise<InjectionConfig> {
    const sig = this.detector.detect(input, context?.fileContext);
    const strategy = this.selectStrategy(sig.type);
    return {
      ...this.adjustParameters(strategy, context ?? {}),
      taskType: sig.type,
      confidence: sig.confidence,
    };
  }
}
```

## 4. 测试设计

| 测试 | 描述 |
|------|------|
| detects coding from "implement feature" | type=coding |
| detects debugging from "fix bug in" | type=debugging |
| detects review from "review this PR" | type=review |
| detects planning from "design architecture" | type=planning |
| falls back to unknown for empty input | type=unknown |
| file context boosts coding detection | test files → coding |
| aggressive strategy includes memory+governance | coding → aggressive |
| minimal strategy for questions | question → minimal/1500 tokens |
| adjustedMaxTokens respects budget | budget=1000 → capped at 1000 |
| urgency=high doubles budget | high urgency → ×1.5 |
