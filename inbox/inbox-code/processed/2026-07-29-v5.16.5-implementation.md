```typescript
static getAllConfigs(): Record<string, ModelConfig> {
  const configs: Record<string, ModelConfig> = {};
  
  for (const [id, profile] of Object.entries(MODEL_PROFILES)) {
    configs[id] = {
      contextWindow: profile.contextWindow,
      compressionThreshold: profile.compressionThreshold,
      effectiveWindowRatio: profile.effectiveWindowRatio || 0.9,
      proactiveThreshold: Math.floor(profile.contextWindow * 0.75)
    };
  }
  
  return configs;
}
```
