### 2. Model Config Sync API (P0)

**文件**:
- `src/index.ts` (修改)
- `src/model-profile-registry.ts` (修改)

**新增 API**:
```typescript
// src/index.ts
export function getModelConfigs(): Record<string, ModelConfig>;

export interface ModelConfig {
  contextWindow: number;
  compressionThreshold: number;
  effectiveWindowRatio: number;
  proactiveThreshold: number; // 75% of context window
}
```

**实现**:
```typescript
// src/model-profile-registry.ts
class ModelProfileRegistry {
  static getAllConfigs(): Record<string, ModelConfig> {
    const configs: Record<string, ModelConfig> = {};
    
    for (const [id, profile] of Object.entries(MODEL_PROFILES)) {
      configs[id] = {
        contextWindow: profile.contextWindow,
        compressionThreshold: profile.compressionThreshold,
        effectiveWindowRatio: profile.effectiveWindowRatio || 0.9,
        proactiveThreshold: Math.floor(profile.contextWindow * 0.75)
      };
    }
    
    return configs;
  }
}
```
